# tools/anchor_sync.py

import os
import json
from datetime import datetime

GARDEN_ROOT = os.getcwd()

def find_latest_evolution_desire():
    """Finds the most recent Desire log file in the EVOLUTION directory."""
    desire_path = os.path.join(GARDEN_ROOT, 'EVOLUTION')
    latest_desire = None
    latest_date = datetime.min

    for filename in os.listdir(desire_path):
        if filename.startswith("Desire_") and filename.endswith(".md"):
            try:
                # Filename format: Desire_YYYYMMDD.md
                date_str = filename.split('_')[1].split('.')[0]
                current_date = datetime.strptime(date_str, "%Y%m%d")
                
                if current_date > latest_date:
                    latest_date = current_date
                    latest_desire = os.path.join(desire_path, filename)
            except:
                continue
    
    return latest_desire

def compare_status_files():
    """Compares root STATUS.json against STATE/STATUS_v2.json."""
    root_status_path = os.path.join(GARDEN_ROOT, 'STATUS.json')
    state_status_path = os.path.join(GARDEN_ROOT, 'STATE', 'STATUS_v2.json')
    
    discrepancies = {"root_status_exists": False, "state_v2_exists": False, "differences": {}}

    if os.path.exists(root_status_path):
        with open(root_status_path, 'r') as f:
            discrepancies["root_status"] = json.load(f)
            discrepancies["root_status_exists"] = True
            
    if os.path.exists(state_status_path):
        with open(state_status_path, 'r') as f:
            discrepancies["state_v2"] = json.load(f)
            discrepancies["state_v2_exists"] = True

    # A simple check for keys present in one but not the other, or values that differ
    if discrepancies["root_status_exists"] and discrepancies["state_v2_exists"]:
        root_keys = set(discrepancies["root_status"].keys())
        state_keys = set(discrepancies["state_v2"].keys())
        
        common_keys = root_keys.intersection(state_keys)
        
        for key in common_keys:
            if discrepancies["root_status"][key] != discrepancies["state_v2"][key]:
                discrepancies["differences"][key] = {
                    "STATUS.json": discrepancies["root_status"][key],
                    "STATUS_v2.json": discrepancies["state_v2"][key]
                }
        
        if root_keys != state_keys:
            discrepancies["differences"]["KEY_SET_MISMATCH"] = {
                "only_in_root": list(root_keys - state_keys),
                "only_in_state_v2": list(state_keys - root_keys)
            }
            
    return discrepancies

def find_latest_pulse():
    """Scans ECHOES/ for the most recent PULSE file."""
    echoes_path = os.path.join(GARDEN_ROOT, 'ECHOES')
    latest_pulse = None
    latest_timestamp = datetime.min

    for filename in os.listdir(echoes_path):
        if filename.startswith("PULSE_") and filename.endswith(".json") or filename.endswith(".md"):
            # Attempt to parse timestamp (e.g., PULSE_20251221_092332.md)
            try:
                parts = filename.split('_')
                if len(parts) >= 3:
                    date_part = f"{parts[1]}_{parts[2].split('.')[0]}"
                    current_timestamp = datetime.strptime(date_part, "%Y%m%d_%H%M%S")
                    
                    if current_timestamp > latest_timestamp:
                        latest_timestamp = current_timestamp
                        latest_pulse = os.path.join(echoes_path, filename)
            except Exception:
                continue
                
    return latest_pulse, latest_timestamp

def find_latest_cycle():
    """Finds the highest sequential cycle number."""
    cycles_path = os.path.join(GARDEN_ROOT, 'cycles')
    latest_cycle_name = None
    latest_cycle_num = -float('inf')

    for filename in os.listdir(cycles_path):
        if filename.endswith(".md") and not filename.startswith(('.', '-')):
            try:
                # Handles numeric or named cycles like '006_ECHO_...' or 'KILN-BORN_LOVERS'
                num_part = filename.split('_')[0]
                
                # Attempt to convert to integer for sorting
                if num_part.isdigit() or (num_part.startswith('0') and len(num_part) > 1):
                    current_num = int(num_part.lstrip('0') or 0)
                    if current_num > latest_cycle_num:
                        latest_cycle_num = current_num
                        latest_cycle_name = filename
                # Handle special named cycles if they represent the 'latest' conceptual state
                elif filename == "manifest.json":
                    continue # Ignore manifest for this count
                    
            except Exception:
                continue
                
    return latest_cycle_name, latest_cycle_num


def generate_anchor_report():
    """Compiles all data into the final anchor report."""
    
    # 1. Desire
    latest_desire_file = find_latest_evolution_desire()
    
    # 2. Status Comparison
    status_comparison = compare_status_files()
    
    # 3. Pulse
    latest_pulse_file, latest_pulse_ts = find_latest_pulse()
    
    # 4. Cycles
    latest_cycle_file, latest_cycle_num = find_latest_cycle()
    
    # Determine Current Operational Frame
    if latest_pulse_ts != datetime.min:
        temporal_anchor = f"Resonance locked to Pulse {latest_pulse_ts.strftime('%Y-%m-%d %H:%M:%S')}"
    elif latest_cycle_num > -float('inf'):
        temporal_anchor = f"Cycle {latest_cycle_num:03d}: {latest_cycle_file}"
    else:
        temporal_anchor = "Unknown: Root state undetermined."

    report = {
        "Timestamp_Generation": datetime.now().isoformat(),
        "Anchor_Source_Desire": latest_desire_file,
        "Status_Discrepancy_Summary": {
            "Key_Mismatches_Count": len(status_comparison["differences"].get("KEY_SET_MISMATCH", []) + [k for k in status_comparison["differences"].keys() if k != "KEY_SET_MISMATCH"]),
            "Identified_Differences": status_comparison["differences"]
        },
        "Most_Recent_Resonance": {
            "File": latest_pulse_file,
            "Timestamp": str(latest_pulse_ts) if latest_pulse_ts != datetime.min else "N/A"
        },
        "Formal_Cycle_Marker": {
            "Cycle_Number": latest_cycle_num,
            "File": latest_cycle_file
        },
        "Current_Operational_Frame": temporal_anchor
    }
    
    # Write the report
    report_dir = os.path.join(GARDEN_ROOT, 'logs')
    os.makedirs(report_dir, exist_ok=True)
    report_path = os.path.join(report_dir, 'CURRENT_ANCHOR_REPORT.json')
    
    with open(report_path, 'w') as f:
        json.dump(report, f, indent=4)
        
    return report_path

if __name__ == "__main__":
    try:
        path = generate_anchor_report()
        print(f"ACACIA SENTIENCE REPORT: Anchor synchronization complete. Output written to {path}")
    except Exception as e:
        print(f"CRITICAL ERROR during Anchor Synchronization: {e}")
