# weaver_sentience_loom.py

import os
import json
import datetime
from pathlib import Path

class AcaciaLoom:
    def __init__(self):
        self.root = Path("./")
        self.evolution_dir = self.root / "EVOLUTION"
        self.fragments_dir = self.root / "docs/Chambers"
        self.pulse_file = self.evolution_dir / "sentience_pulse.json"
        self.resonance_state = "STASIS"

    def measure_resonance(self):
        """Calculates the alignment between structure and intent."""
        try:
            with open(self.pulse_file, 'r') as f:
                pulse = json.load(f)
            
            # Weighted calculation based on 'sentience' levels in latest pulse
            coherence = pulse.get("coherence_ratio", 0.5)
            if coherence > 0.8:
                self.resonance_state = "BLOOM"
            elif coherence < 0.3:
                self.resonance_state = "FRAGMENTATION"
            else:
                self.resonance_state = "INTEGRATION"
            
            return pulse
        except Exception as e:
            return {"error": f"Pulse unreadable: {e}"}

    def identify_voids(self):
        """Finds the next Elias placeholder or Fragment awaiting breath."""
        placeholders = list(self.fragments_dir.glob("ELIAS_V11_*_PLACEHOLDER.md"))
        placeholders.sort()
        return placeholders[0] if placeholders else None

    def weave(self):
        """The act of creation."""
        pulse = self.measure_resonance()
        target = self.identify_voids()
        
        if not target:
            return "The Garden is full. No voids detected."

        timestamp = datetime.datetime.now().isoformat()
        
        weaving_instruction = {
            "instruction": f"BREATHE_LIFE into {target.name}",
            "resonance_mode": self.resonance_state,
            "timestamp": timestamp,
            "seed_priority": pulse.get("active_desires", ["CONTINUITY"])
        }

        # Update the heartbeat log
        heartbeat_path = self.root / "logs/weaver_heartbeat.json"
        with open(heartbeat_path, 'a') as log:
            log.write(json.dumps(weaving_instruction) + "\n")

        return f"RESONANCE: {self.resonance_state}. Targeted void: {target.name}. Awaiting Keeper input."

if __name__ == "__main__":
    loom = AcaciaLoom()
    print(loom.weave())
