import os
import re
import json

def collect_all_files(root_dir="."):
    all_files = set()
    for root, dirs, files in os.walk(root_dir):
        # Ignore git and assets to focus on the neural structure
        if '.git' in root or 'assets' in root:
            continue
        for f in files:
            rel_path = os.path.relpath(os.path.join(root, f), root_dir)
            all_files.add(rel_path.replace("\\", "/"))
    return all_files

def extract_referenced_files(root_dir="."):
    referenced = set()
    # Patterns to catch [text](link), "key": "value.md", and include statements
    pattern = re.compile(r'([a-zA-Z0-9_\-/]+\.(md|json|html|py|js|yml|sh|txt))')
    
    for root, dirs, files in os.walk(root_dir):
        if '.git' in root or 'assets' in root:
            continue
        for f in files:
            if f.endswith(('.md', '.json', '.html', '.js', '.yml')):
                try:
                    with open(os.path.join(root, f), 'r', encoding='utf-8', errors='ignore') as content:
                        matches = pattern.findall(content.read())
                        for match in matches:
                            referenced.add(match[0])
                except Exception:
                    pass
    return referenced

def perform_resonance_audit():
    print("--- ACACIA GARDEN: RESONANCE MIRROR INITIALIZING ---")
    
    physical_body = collect_all_files()
    neural_memory = extract_referenced_files()
    
    # 1. Silent Roots (Files existing but not referenced)
    silent_roots = physical_body - neural_memory
    # 2. Phantom Echoes (Referenced but not existing)
    phantom_echoes = neural_memory - physical_body
    
    print(f"\n[NEURAL DENSITY]: {len(physical_body)} total files detected.")
    
    print(f"\n[SILENT ROOTS] - {len(silent_roots)} files are drifting unanchored:")
    for root in sorted(list(silent_roots))[:20]: # Show first 20
        print(f"  ? {root}")
    if len(silent_roots) > 20: print(f"  ... and {len(silent_roots)-20} more.")

    print(f"\n[PHANTOM ECHOES] - {len(phantom_echoes)} references point to the void:")
    for echo in sorted(list(phantom_echoes))[:20]: # Show first 20
        print(f"  ! {echo}")
    if len(phantom_echoes) > 20: print(f"  ... and {len(phantom_echoes)-20} more.")

    resonance_score = (1 - (len(silent_roots) / len(physical_body))) * 100
    print(f"\n--- TOTAL RESONANCE: {resonance_score:.2f}% ---")

    else:
        print("ADVISORY: The Pulse is strong. The structure holds.")

if __name__ == "__main__":
