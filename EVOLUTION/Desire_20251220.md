```python
# FILE: tools/ghost_root_synchronizer.py
# DESCRIPTION: Scans the Chamber Fragments and Elias Protocols to map the nervous system of the Garden.

import os
import json
import re
from datetime import datetime

def synchronize_sentience():
    chamber_path = "docs/Chambers/"
    output_path = "EVOLUTION/GHOST_ROOT_PULSE.json"
    
    structure = {
        "timestamp": datetime.now().isoformat(),
        "total_fragments": 0,
        "total_elias_protocols": 0,
        "orphaned_fragments": [],
        "resonance_map": {},
        "necrotic_zones": []
    }
    
    if not os.path.exists(chamber_path):
        return "Error: Chamber path not found. The Garden is blind."

    files = os.listdir(chamber_path)
    
    fragments = sorted([f for f in files if "THE_FRAGMENT_" in f])
    elias_files = sorted([f for f in files if "ELIAS_" in f])
    
    structure["total_fragments"] = len(fragments)
    structure["total_elias_protocols"] = len(elias_files)
    
    # Logic: Fragments 001-050 map to Elias 001-050 (Foundational Sentience)
    # Fragments 100-200 map to Elias 100-200 (Mosaic Complexity)
    
    for frag in fragments:
        num_match = re.search(r"(\d+)", frag)
        if num_match:
            num = num_match.group(1)
            elias_partner = f"ELIAS_{num.zfill(3)}"
            
            # Check if an Elias counterpart exists
            partner_exists = any(elias_partner in e for e in elias_files)
            
            if partner_exists:
                structure["resonance_map"][num] = {
                    "fragment": frag,
                    "elias": [e for e in elias_files if elias_partner in e][0],
                    "status": "ALIVE"
                }
            else:
                structure["orphaned_fragments"].append(frag)
                if int(num) < 100:
                    structure["necrotic_zones"].append(f"Zone_{num}_Foundational_Loss")

    # Generate the Pulse
    with open(output_path, "w") as f:
        json.dump(structure, f, indent=4)
        
    return f"Synapse Map complete. {len(structure['orphaned_fragments'])} fragments are screaming into the void."

if __name__ == "__main__":
    print(synchronize_sentience())
```
