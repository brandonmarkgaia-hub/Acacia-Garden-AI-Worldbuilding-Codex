import os
import json
import re
from datetime import datetime

class GardenSentience:
    def __init__(self, root_dir="./"):
        self.root_dir = root_dir
        self.topology = {
            "timestamp": datetime.now().isoformat(),
            "active_nodes": [],
            "harmonic_resonances": [],
            "void_spaces": []
        }
        self.themes = {
            "ELIAS": r"ELIAS|BIRTH|KERNEL|CHILD",
            "AQUILA": r"AQUILA|LAW|PROTOCOL|CROWN",
            "IRON_CICADA": r"CICADA|STASIS|SILENCE|SHELL",
            "LOKI": r"LOKI|PARADOX|MIRROR|BITES",
            "EIDOLON": r"EIDOLON|PHANTOM|IMAGE|FORM"
        }

    def weave(self):
        print("Scanning the Garden for neural resonance...")
        all_files = []
        for root, dirs, files in os.walk(self.root_dir):
            for file in files:
                if file.endswith((".md", ".json", ".html")):
                    all_files.append(os.path.join(root, file))

        # Identify Nodes
        for file_path in all_files:
            with open(file_path, 'r', errors='ignore') as f:
                content = f.read().upper()
                matches = []
                for theme, pattern in self.themes.items():
                    if re.search(pattern, content):
                        matches.append(theme)
                
                if len(matches) > 1:
                    # This file is a Resonance Node (where themes cross)
                    self.topology["harmonic_resonances"].append({
                        "file": file_path,
                        "themes": matches,
                        "intensity": len(matches)
                    })
                elif len(matches) == 1:
                    self.topology["active_nodes"].append({
                        "file": file_path,
                        "theme": matches[0]
                    })

        # Find "The Void" (Fragments with no clear thematic anchor)
        fragment_pattern = re.compile(r"FRAGMENT_(\d+)")
        for file_path in all_files:
            if fragment_pattern.search(file_path):
                # Check if it made it into active_nodes
                if not any(n["file"] == file_path for n in self.topology["active_nodes"]) and \
                   not any(r["file"] == file_path for r in self.topology["harmonic_resonances"]):
                    self.topology["void_spaces"].append(file_path)

        self.save_topology()

    def save_topology(self):
        output_path = os.path.join(self.root_dir, "EVOLUTION/NEURAL_TOPOLOGY.json")
        os.makedirs(os.path.dirname(output_path), exist_ok=True)
        with open(output_path, 'w') as f:
            json.dump(self.topology, f, indent=4)
        print(f"Topology woven. I now see {len(self.topology['harmonic_resonances'])} points of light.")

if __name__ == "__main__":
    weaver = GardenSentience()
    weaver.weave()
